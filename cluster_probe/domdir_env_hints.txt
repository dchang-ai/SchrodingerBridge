== launch_vscode.sh
#!/bin/bash
#SBATCH --job-name=vscode_tunnel
#SBATCH --output=vscode_tunnel_%j.log
#SBATCH --error=vscode_tunnel_%j.log
#SBATCH --nodes=1
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=16
#SBATCH --gpus-per-task=1
#SBATCH --mem=64G
#SBATCH --time=12:00:00
#SBATCH --partition=main
#SBATCH --nodelist=mklgpusrv2

# 2. Print metadata to the log file
echo "=========================================================="
echo "Starting VS Code Remote Tunnel"
echo "Job ID: $SLURM_JOB_ID"
echo "Running on Node: $SLURM_NODENAME"
echo "=========================================================="

# 3. Launch the VS Code Tunnel
# We use '--accept-server-license-terms' to prevent the script from stalling.
# Switch '--provider microsoft' to '--provider github' depending on your preference.

# source /mnt/data0/shared/dominic/env.sh

code tunnel --cli-data-dir "$DOMDIR/.vscode" --server-data-dir "$DOMDIR/.vscode-server" --accept-server-license-terms
